cmake_minimum_required(VERSION 3.20)
project(sfa_developer_console C)
set(CMAKE_C_STANDARD 99)
if(APPLE AND NOT CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
 message(FATAL_ERROR "Foxhollow macOS target is Apple Silicon only.")
endif()
if(WIN32)
 set(PLATFORM_SRC src/platform/windows.c)
 set(FH_PLATFORM windows-amd64)
elseif(APPLE)
 set(PLATFORM_SRC src/platform/sdl_unix.c)
 set(FH_PLATFORM macos-arm64)
elseif(UNIX)
 set(PLATFORM_SRC src/platform/sdl_unix.c)
 set(FH_PLATFORM linux-amd64)
else()
 message(FATAL_ERROR "Unsupported platform")
endif()
add_library(sfa_developer_console SHARED src/mod.c src/console.c ${PLATFORM_SRC})
target_include_directories(sfa_developer_console PRIVATE include src)
if(UNIX AND NOT APPLE)
 target_link_libraries(sfa_developer_console PRIVATE dl)
endif()
set_target_properties(sfa_developer_console PROPERTIES PREFIX "" OUTPUT_NAME "mod")
set(OUT_ROOT "${CMAKE_BINARY_DIR}/sfa-developer-console")
set(OUT_LIB "${OUT_ROOT}/lib/${FH_PLATFORM}")
add_custom_command(TARGET sfa_developer_console POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E make_directory "${OUT_LIB}"
 COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:sfa_developer_console>" "${OUT_LIB}/$<TARGET_FILE_NAME:sfa_developer_console>"
 COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/mod.json" "${OUT_ROOT}/mod.json")
